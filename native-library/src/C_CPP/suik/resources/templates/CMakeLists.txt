cmake_minimum_required(VERSION 3.22)
project(sui_project LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(math src/math.cpp src/c_math.c)
target_include_directories(math PUBLIC include)
add_executable(app src/main.cpp)
target_link_libraries(app PRIVATE math)
enable_testing()
add_executable(math_test tests/math_test.cpp)
target_link_libraries(math_test PRIVATE math)
add_test(NAME arithmetic COMMAND math_test)
